def replay(self) -> None:
    if not get_moe_a2a_backend().is_deepep():
        return

    assert self._captured_deepep_mode is not None

    # Diagnostic A/B for NPU graph replay:
    # avoid cleaning the LL buffer immediately before replaying
    # a graph captured with LOW_LATENCY DeepEP.
    if _is_npu and self._captured_deepep_mode.is_low_latency():
        state = DeepEPBuffer._state()

        print(
            f"[DEEPEP-GRAPH] replay: "
            f"current_mode={state.dispatch_mode}, "
            f"captured_mode={self._captured_deepep_mode}",
            flush=True,
        )

        if state.dispatch_mode == DeepEPDispatchMode.NORMAL:
            print(
                "[DEEPEP-GRAPH] NORMAL -> LOW_LATENCY "
                "WITHOUT clean_low_latency_buffer [A/B TEST]",
                flush=True,
            )
            state.dispatch_mode = DeepEPDispatchMode.LOW_LATENCY
            return

    DeepEPBuffer.set_dispatch_mode(self._captured_deepep_mode)

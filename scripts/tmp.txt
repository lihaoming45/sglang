# Diagnostic A/B for NPU compact graph padding.
# Keep padding expert ids legal for DeepEP LL, but make their routing weight zero.
if num_token_non_padded is not None:
    row_idx = torch.arange(
        topk_ids.shape[0],
        device=topk_ids.device,
        dtype=num_token_non_padded.dtype,
    )

    valid_rows = (row_idx < num_token_non_padded).unsqueeze(-1)

    # IMPORTANT:
    # Do NOT use -1 on NPU DeepEP LL.
    # Expert 0 is a legal placeholder for graph-padding rows.
    topk_ids = torch.where(
        valid_rows,
        topk_ids,
        torch.zeros_like(topk_ids),
    )

    topk_weights = torch.where(
        valid_rows,
        topk_weights,
        torch.zeros_like(topk_weights),
    )

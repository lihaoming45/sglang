# Compact ragged verify: input_ids is padded to graph_num_tokens, but
# num_token_non_padded must describe only the real verify tokens.
if self.ragged_verify_layout is not None and not batch.forward_mode.is_idle():
    layout = self.ragged_verify_layout

    if layout.verify_lens_cpu is not None:
        real_num_tokens = int(sum(layout.verify_lens_cpu))
    else:
        real_num_tokens = int(layout.verify_lens.sum().item())

    verify_forward_batch.global_num_token_non_padded_cpu = real_num_tokens

    if verify_forward_batch.global_num_token_non_padded is not None:
        verify_forward_batch.global_num_token_non_padded.fill_(real_num_tokens)

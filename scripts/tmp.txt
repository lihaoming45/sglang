def replay(
    self,
    shape_key: ShapeKey,
    static_forward_batch: ForwardBatch,
    **kwargs,
):
    rank = (
        torch.distributed.get_rank()
        if torch.distributed.is_initialized()
        else -1
    )

    print(
        f"[REPLAY-1] rank={rank} BEFORE_SYNC "
        f"shape_key={shape_key}",
        flush=True,
    )

    torch.npu.synchronize()

    print(
        f"[REPLAY-2] rank={rank} BEFORE_SYNC_DONE "
        f"shape_key={shape_key}",
        flush=True,
    )

    self._graphs[shape_key].replay()

    print(
        f"[REPLAY-3] rank={rank} REPLAY_RETURNED "
        f"shape_key={shape_key}",
        flush=True,
    )

    torch.npu.synchronize()

    print(
        f"[REPLAY-4] rank={rank} AFTER_SYNC_DONE "
        f"shape_key={shape_key}",
        flush=True,
    )

    return self._outputs[shape_key]
